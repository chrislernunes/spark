cmake_minimum_required(VERSION 3.16)
project(spark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(spark_test tests/test_main.cpp)
target_include_directories(spark_test PRIVATE include)
add_executable(spark_bench benchmarks/bench.cpp)
target_include_directories(spark_bench PRIVATE include)
add_executable(sparkctl apps/sparkctl.cpp)
target_include_directories(sparkctl PRIVATE include)
enable_testing()
add_test(NAME spark_test COMMAND spark_test WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
